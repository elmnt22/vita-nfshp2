cmake_minimum_required(VERSION 3.20)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(NFSHP2ArchiveUtility LANGUAGES CXX)
endif()

add_executable(NFSHP2Unpack ArchiveTool.cpp Sha256.hpp KnownNames.inc LegacyPalNames.inc)
target_compile_features(NFSHP2Unpack PRIVATE cxx_std_17)
set_target_properties(NFSHP2Unpack PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(NFSHP2Unpack PRIVATE /W4 /WX /utf-8)
    set_property(TARGET NFSHP2Unpack PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(NFSHP2Unpack PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(WIN32)
    target_compile_definitions(NFSHP2Unpack PRIVATE UNICODE _UNICODE)
    target_link_libraries(NFSHP2Unpack PRIVATE comdlg32 user32)
    if(MINGW)
        target_link_options(NFSHP2Unpack PRIVATE -municode -static)
    endif()
endif()
